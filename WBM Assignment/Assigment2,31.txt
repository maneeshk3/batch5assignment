select a. "user" from TableA a
where a. "user" Not in (
select um."Source"
from UserMappingA um);

-----Query 2: Update the user column of Table A with values from Target column of User mapping where user matches with Source.
UPDATE TableA a
SET a."user" = (
    SELECT um."Target"
    FROM UserMappingA um
    WHERE a."user" = um."Source"
);
